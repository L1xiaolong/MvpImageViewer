qt_add_library(mvpview_render_backend STATIC
    bayer_render_parameters.h
    bayer_render_parameters.cpp
    yuv_render_parameters.h
    yuv_render_parameters.cpp
)

target_include_directories(mvpview_render_backend PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(mvpview_render_backend
    PUBLIC mvpview_core
    PRIVATE Qt6::GuiPrivate Qt6::ShaderTools
)

qt_add_shaders(mvpview_render_backend "mvpview_image_shaders"
    PREFIX "/shaders"
    BASE "shaders"
    FILES
        shaders/image.vert
        shaders/image.frag
        shaders/yuv.frag
        shaders/yuv_compare.frag
)

# Packed Bayer decoding needs integer bit operations and texelFetch, which are not
# available in the legacy GLSL 100 ES/120 variants generated by default.
qt_add_shaders(mvpview_render_backend "mvpview_bayer_shaders"
    PREFIX "/shaders"
    BASE "shaders"
    GLSL "300 es,140,330"
    FILES
        shaders/bayer.frag
        shaders/bayer_compare.frag
        shaders/qml_side.frag
)

mvpview_enable_warnings(mvpview_render_backend)
